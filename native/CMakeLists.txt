# Deprecated aliases from before the renderer was renamed; existing build
# directories keep working until they are reconfigured with the new names.
foreach(alias_entry "SKY_BBPORT_ROOT;SKY_RENDERER_ROOT;PATH" "SKY_BBGPU_LIBRARY;SKY_RENDERER_LIBRARY;FILEPATH")
  list(GET alias_entry 0 old_name)
  list(GET alias_entry 1 new_name)
  list(GET alias_entry 2 cache_type)
  if(${old_name} AND NOT ${new_name})
    message(DEPRECATION "${old_name} is deprecated; use ${new_name}")
    get_property(cache_help CACHE ${new_name} PROPERTY HELPSTRING)
    set(${new_name} "${${old_name}}" CACHE ${cache_type} "${cache_help}" FORCE)
  endif()
endforeach()
if(SKY_RENDERER_ROOT OR SKY_RENDERER_LIBRARY)
  if(NOT EXISTS "${SKY_RENDERER_ROOT}/src/runtime_memory.c" OR NOT EXISTS "${SKY_RENDERER_LIBRARY}")
    message(FATAL_ERROR "Both the pinned renderer source and matching renderer library are required")
  endif()
  target_sources(skylanders-native-host PRIVATE pc_overlay_bridge.cpp renderer_adapter.cpp)
  enable_language(C)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(SkySDL3 REQUIRED IMPORTED_TARGET sdl3)
  file(GLOB SKY_ATRAC9_SOURCES CONFIGURE_DEPENDS "${SKY_RENDERER_ROOT}/third_party/LibAtrac9/C/src/*.c")
  if(NOT SKY_ATRAC9_SOURCES)
    message(FATAL_ERROR "The pinned MIT LibAtrac9 dependency is required")
  endif()
  set(SKY_PRIVATE_RUNTIME_SOURCES "${SKY_RENDERER_ROOT}/src/runtime_memory.c"
      "${SKY_RENDERER_ROOT}/src/runtime_ajm.c" "${SKY_RENDERER_ROOT}/src/runtime_audio.c" ${SKY_ATRAC9_SOURCES})
  target_sources(skylanders-native-host PRIVATE ${SKY_PRIVATE_RUNTIME_SOURCES})
  target_include_directories(skylanders-native-host PRIVATE "${SKY_RENDERER_ROOT}" "${SKY_RENDERER_ROOT}/src" "${SKY_RENDERER_ROOT}/gpu")
  target_compile_definitions(skylanders-native-host PRIVATE SKY_RENDERER=1)
  target_link_libraries(skylanders-native-host PRIVATE "${SKY_RENDERER_LIBRARY}" PkgConfig::SkySDL3)
  target_link_options(skylanders-native-host PRIVATE -Wl,--export-dynamic)
endif()
